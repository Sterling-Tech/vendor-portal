const daysAgo = (days: number) => {

const formatCurrency = (value: number | string) =>
    new Intl.NumberFormat('en-NG', {
        style: 'currency',
        currency: 'NGN',
        maximumFractionDigits: 2,
    }).format(Number(value || 0));

const formatDate = (value: string) => {
    const date = new Date(value);
    return Number.isNaN(date.getTime()) ? new Date().toLocaleDateString() : date.toLocaleDateString();
};

const slugify = (value: string) =>
    value
        .trim()
        .toLowerCase()
        .replace(/[^a-z0-9]+/g, '-')
        .replace(/(^-|-$)/g, '');

const statusColor = (status: string): ChipColor => {
    switch (status) {
        case 'active':
            return 'success';
        case 'suspended':
            return 'warning';
        case 'invited':
            return 'info';
        case 'review':
            return 'secondary';
        default:
            return 'default';
    }
};

const levelColor = (level: OrgLevel): ChipColor => {
    switch (level) {
        case 'vendor':
            return 'primary';
        case 'sub_vendor':
            return 'success';
        case 'agent':
            return 'warning';
        case 'cashier':
            return 'info';
        default:
            return 'default';
    }
};

const levelLabel = (level: OrgLevel) => LEVEL_OPTIONS.find((option) => option.value === level)?.label || level;

const nextChildLevel = (level: OrgLevel): OrgLevel => {
    if (level === 'vendor') return 'sub_vendor';
    if (level === 'sub_vendor') return 'agent';
    return 'cashier';
};

const countChildren = (nodes: VendorOrg[] = []): number =>
    nodes.reduce((total, node) => total + 1 + countChildren(node.children || []), 0);

const flattenTree = (nodes: VendorOrg[], path: string[] = [], depth = 0): FlatOrg[] =>
    nodes.flatMap((node) => [
        { ...node, path, depth, childCount: countChildren(node.children || []) },
        ...flattenTree(node.children || [], [...path, node.name], depth + 1),
    ]);

const findOrg = (nodes: VendorOrg[], id: string | null): VendorOrg | null => {
    if (!id) return null;
    for (const node of nodes) {
        if (node.id === id) return node;
        const child = findOrg(node.children || [], id);
        if (child) return child;
    }
    return null;
};

const addOrgToTree = (nodes: VendorOrg[], parentId: string | null, newOrg: VendorOrg): VendorOrg[] => {
    if (!parentId) return [newOrg, ...nodes];

    return nodes.map((node) => {
        if (node.id === parentId) {
            return { ...node, children: [newOrg, ...(node.children || [])] };
        }
        return { ...node, children: addOrgToTree(node.children || [], parentId, newOrg) };
    });
};

const updateOrgInTree = (nodes: VendorOrg[], id: string, patch: Partial<VendorOrg>): VendorOrg[] =>
    nodes.map((node) => {
        if (node.id === id) return { ...node, ...patch };
        return { ...node, children: updateOrgInTree(node.children || [], id, patch) };
    });

const removeOrgFromTree = (nodes: VendorOrg[], id: string): VendorOrg[] =>
    nodes
        .filter((node) => node.id !== id)
        .map((node) => ({ ...node, children: removeOrgFromTree(node.children || [], id) }));

                            ) : (

                            )}
                            <ListItemText
                                primary={node.name}
                                secondary={
                                    <Stack direction="row" spacing={1} mt={0.5} flexWrap="wrap">
                                        <Chip label={levelLabel(node.level)} size="small" color={levelColor(node.level)} variant="outlined" />
                                        <Chip label={node.status} size="small" color={statusColor(node.status)} />
                                    </Stack>
                                }
                            />
                        </ListItemButton>
                        {hasChildren && isExpanded && renderTree(node.children || [], depth + 1)}
                    </Box>
                );
            })}
        </List>
    );

    return (
        <Box
            p={{ xs: 2, md: 3 }}
            sx={{
                '@keyframes tableSlideIn': {
                    from: { opacity: 0, transform: 'translateX(-28px)' },
                    to: { opacity: 1, transform: 'translateX(0)' },
                },
                '@keyframes detailSlideIn': {
                    from: { opacity: 0, transform: 'translateX(36px)' },
                    to: { opacity: 1, transform: 'translateX(0)' },
                },
            }}
        >
            <Stack direction={{ xs: 'column', md: 'row' }} justifyContent="space-between" alignItems={{ xs: 'stretch', md: 'center' }} spacing={2} mb={3}>
                <Box>
                    <Typography variant="h4">Vendor Hierarchy</Typography>
                    <Typography variant="body2" color="text.secondary">
                        {selectedTenant.name} - updated {lastUpdated.toLocaleTimeString()}
                    </Typography>
                </Box>
                <Stack direction="row" spacing={1}>
                    <Button variant="outlined" startIcon={<Refresh />} onClick={() => setLastUpdated(new Date())}>
                        Refresh
                    </Button>
                    <Button variant="contained" startIcon={<Add />} onClick={() => openCreateDialog(null, 'vendor')}>
                        Add Vendor Admin
                    </Button>
                </Stack>
            </Stack>

            <Grid container spacing={2} mb={3}>
                <Grid item xs={12} md={3}>
                    <StatCard title="Today’s Processed Volume" value={formatCurrency(stats.total)} icon={<Business />} color="primary" />
                </Grid>
                <Grid item xs={12} md={3}>
                    <StatCard title="Active" value={formatNumber(stats.active)} icon={<CheckCircle />} color="success" />
                </Grid>
                <Grid item xs={12} md={3}>
                    <StatCard title="Total Balance" value={formatCurrency(stats.totalBalance)} icon={<AccountBalance />} color="info" />
                </Grid>
                <Grid item xs={12} md={3}>
                    <StatCard title="Transactions" value={formatNumber(stats.totalTransactions)} icon={<Receipt />} color="warning" />
                </Grid>
            </Grid>

            {!selectedOrg ? (
                <Box key="vendor-table" sx={{ animation: 'tableSlideIn 340ms ease' }}>
                    <Card sx={{ mb: 3, borderRadius: 1 }}>
                        <CardContent>
                            <Grid container spacing={2} alignItems="center">
                                <Grid item xs={12} md={4}>
                                    <TextField
                                        fullWidth
                                        size="small"
                                        placeholder="Search name, location, email"
                                        value={searchTerm}
                                        onChange={(event) => {
                                            setSearchTerm(event.target.value);
                                            setPage(0);
                                        }}
                                        InputProps={{
                                            startAdornment: (
                                                <InputAdornment position="start">
                                                    <Search fontSize="small" />
                                                </InputAdornment>
                                            ),
                                        }}
                                    />
                                </Grid>
                                <Grid item xs={12} md={2}>
                                    <FormControl fullWidth size="small">
                                        <InputLabel>Level</InputLabel>
                                        <Select value={levelFilter} label="Level" onChange={(event) => { setLevelFilter(event.target.value as OrgLevel | 'all'); setPage(0); }}>
                                            <MenuItem value="all">All Levels</MenuItem>
                                            {LEVEL_OPTIONS.map((option) => (
                                                <MenuItem key={option.value} value={option.value}>{option.label}</MenuItem>
                                            ))}
                                        </Select>
                                    </FormControl>
                                </Grid>
                                <Grid item xs={12} md={2}>
                                    <FormControl fullWidth size="small">
                                        <InputLabel>Status</InputLabel>
                                        <Select value={statusFilter} label="Status" onChange={(event) => { setStatusFilter(event.target.value as OrgStatus | 'all'); setPage(0); }}>
                                            <MenuItem value="all">All Status</MenuItem>
                                            {STATUS_OPTIONS.map((option) => (
                                                <MenuItem key={option.value} value={option.value}>{option.label}</MenuItem>
                                            ))}
                                        </Select>
                                    </FormControl>
                                </Grid>
                                <Grid item xs={12} md={2}>
                                    <TextField
                                        fullWidth
                                        size="small"
                                        type="date"
                                        label="Start Date"
                                        value={startDate}
                                        onChange={(event) => { setStartDate(event.target.value); setPage(0); }}
                                        InputLabelProps={{ shrink: true }}
                                    />
                                </Grid>
                                <Grid item xs={12} md={2}>
                                    <TextField
                                        fullWidth
                                        size="small"
                                        type="date"
                                        label="End Date"
                                        value={endDate}
                                        onChange={(event) => { setEndDate(event.target.value); setPage(0); }}
                                        InputLabelProps={{ shrink: true }}
                                    />
                                </Grid>
                                <Grid item xs={12}>
                                    <Stack direction="row" spacing={1} flexWrap="wrap" alignItems="center">
                                        <Chip label={${filteredOrgs.length} filtered} color="primary" variant="outlined" />
                                        <Chip label={${stats.vendors} vendors} size="small" />
                                        <Chip label={${stats.subVendors} sub-vendors} size="small" />
                                        <Chip label={${stats.agents} agents} size="small" />
                                        <Chip label={${stats.cashiers} cashiers} size="small" />
                                        <Button size="small" onClick={resetFilters}>Reset Filters</Button>
                                    </Stack>
                                </Grid>
                            </Grid>
                        </CardContent>
                    </Card>

                    <Paper variant="outlined" sx={{ p: 2, borderRadius: 1 }}>
                        <Stack direction={{ xs: 'column', md: 'row' }} justifyContent="space-between" spacing={2} mb={2}>
                            <Stack direction="row" spacing={1.5} alignItems="center">
                                <Avatar sx={{ bgcolor: 'primary.light', color: 'primary.main' }}><Domain /></Avatar>
                                <Box>
                                    <Typography variant="h6">Recent Vendors</Typography>
                                    <Typography variant="body2" color="text.secondary">
                                        Click any row to slide into the full vendor management details.
                                    </Typography>
                                </Box>
                            </Stack>

                            {/* <FormControl size="small" sx={{ minWidth: 260 }}>
                                <InputLabel>Tenant</InputLabel>
                                <Select
                                    value={tenantId}
                                    label="Tenant"
                                    onChange={(event) => setTenantId(event.target.value)}
                                    onOpen={() => setTenantSearchTerm('')}
                                >
                                    {filteredTenants.map((tenant) => (
                                        <MenuItem key={tenant.id} value={tenant.id}>
                                            {tenant.name}
                                        </MenuItem>
                                    ))}
                                </Select>
                            </FormControl> */}
                        </Stack>

                        <Box sx={{ overflowX: 'auto' }}>
                            <Table size="small">
                                <TableHead>
                                    <TableRow>
                                        <TableCell>Organisation</TableCell>
                                        <TableCell>Level</TableCell>
                                        <TableCell>Status</TableCell>
                                        <TableCell>Location</TableCell>
                                        <TableCell>Created</TableCell>
                                        <TableCell align="right">Balance</TableCell>
                                        <TableCell align="right">Transactions</TableCell>
                                        <TableCell align="right">Children</TableCell>
                                        <TableCell align="right">Action</TableCell>
                                    </TableRow>
                                </TableHead>
                                <TableBody>
                                    {paginatedOrgs.map((org) => (
                                        <TableRow key={org.id} hover sx={{ cursor: 'pointer' }} onClick={() => handleSelectOrg(org.id)}>
                                            <TableCell>
                                                <Stack spacing={0.25}>
                                                    <Typography variant="subtitle2">{org.name}</Typography>
                                                    <Typography variant="caption" color="text.secondary">
                                                        {org.path.length ? org.path.join(' / ') : 'Top level'}
                                                    </Typography>
                                                </Stack>
                                            </TableCell>
                                            <TableCell><Chip label={levelLabel(org.level)} size="small" color={levelColor(org.level)} variant="outlined" /></TableCell>
                                            <TableCell><Chip label={org.status} size="small" color={statusColor(org.status)} /></TableCell>
                                            <TableCell>{org.location}</TableCell>
                                            <TableCell>{formatDate(org.created_at)}</TableCell>
                                            <TableCell align="right">{formatCurrency(org.wallet_balance)}</TableCell>
                                            <TableCell align="right">{formatNumber(org.transactions)}</TableCell>
                                            <TableCell align="right">{formatNumber(org.childCount)}</TableCell>
                                            <TableCell align="right">
                                                <Button
                                                    size="small"
                                                    endIcon={<ArrowForward />}
                                                    onClick={(event) => {
                                                        event.stopPropagation();
                                                        handleSelectOrg(org.id);
                                                    }}
                                                >
                                                    View
                                                </Button>
                                            </TableCell>
                                        </TableRow>
                                    ))}
                                </TableBody>
                            </Table>
                        </Box>

                        <TablePagination
                            component="div"
                            count={filteredOrgs.length}
                            page={page}
                            onPageChange={(_, nextPage) => setPage(nextPage)}
                            rowsPerPage={rowsPerPage}
                            onRowsPerPageChange={(event) => {
                                setRowsPerPage(Number(event.target.value));
                                setPage(0);
                            }}
                            rowsPerPageOptions={[5, 8, 12, 20]}
                        />
                    </Paper>
                </Box>
            ) : (
                <Box id="vendor-detail-panel" key={selectedOrg.id} sx={{ animation: 'detailSlideIn 380ms ease' }}>
                    <Stack direction={{ xs: 'column', md: 'row' }} justifyContent="space-between" alignItems={{ xs: 'stretch', md: 'center' }} spacing={2} mb={2}>
                        <Box>
                            <Typography variant="h6">Selected Vendor Details</Typography>
                            <Typography variant="body2" color="text.secondary">{selectedOrg.name}</Typography>
                        </Box>
                        <Stack direction="row" spacing={1}>
                            <Button variant="outlined" startIcon={<ArrowBack />} onClick={() => setSelectedOrgId(null)}>
                                Back to Recent Vendors
                            </Button>
                            <Button variant="outlined" startIcon={<Add />} onClick={() => openCreateDialog(selectedOrg.id, nextChildLevel(selectedOrg.level))}>
                                Add Child
                            </Button>
                        </Stack>
                    </Stack>

                    <Grid container spacing={2} mb={3}>
                        <Grid item xs={12} md={3}>
                            <StatCard title="Wallet Balance" value={formatCurrency(selectedOrg.wallet_balance)} icon={<AccountBalance />} color="success" />
                        </Grid>
                        <Grid item xs={12} md={3}>
                            <StatCard title="Transactions" value={formatNumber(selectedOrg.transactions)} icon={<Receipt />} color="warning" />
                        </Grid>
                        <Grid item xs={12} md={3}>
                            <StatCard title="Users" value={formatNumber(selectedOrg.users)} icon={<People />} color="info" />
                        </Grid>
                        <Grid item xs={12} md={3}>
                            <StatCard title="Child Organisations" value={formatNumber(countChildren(selectedOrg.children || []))} icon={<Business />} color="primary" />
                        </Grid>
                    </Grid>

                    <Grid container spacing={3}>
                        <Grid item xs={12} md={7}>
                            <Stack spacing={3}>
                                <Card sx={{ borderRadius: 1 }}>
                                    <CardContent>
                                        <Stack direction={{ xs: 'column', md: 'row' }} justifyContent="space-between" spacing={2} mb={2}>
                                            <Stack spacing={0.5}>
                                                <Typography variant="h5">{selectedOrg.name}</Typography>
                                                <Typography color="text.secondary">{selectedOrg.manager_email}</Typography>
                                                <Stack direction="row" spacing={1} flexWrap="wrap">
                                                    <Chip label={levelLabel(selectedOrg.level)} color={levelColor(selectedOrg.level)} />
                                                    <Chip label={selectedOrg.status} color={statusColor(selectedOrg.status)} />
                                                    <Chip label={selectedOrg.location} variant="outlined" />
                                                </Stack>
                                            </Stack>
                                            <Stack direction="row" spacing={1}>
                                                <Button variant="outlined" startIcon={<Edit />} onClick={openEditDialog}>Edit</Button>
                                                <Button variant="outlined" color="error" startIcon={<Delete />} onClick={() => setDeleteDialogOpen(true)}>Delete</Button>
                                            </Stack>
                                        </Stack>
                                        <Grid container spacing={2}>
                                            <Grid item xs={12} md={6}>
                                                <Typography variant="body2" color="text.secondary">Slug</Typography>
                                                <Typography fontFamily="monospace">{selectedOrg.slug}</Typography>
                                            </Grid>
                                            <Grid item xs={12} md={6}>
                                                <Typography variant="body2" color="text.secondary">Created</Typography>
                                                <Typography>{formatDate(selectedOrg.created_at)}</Typography>
                                            </Grid>
                                            <Grid item xs={12}>
                                                <Button size="small" onClick={() => navigate(/tenants/vendors/${selectedOrg.id})}>
                                                    View Full Details
                                                </Button>
                                            </Grid>
                                        </Grid>
                                    </CardContent>
                                </Card>

                                <Card sx={{ borderRadius: 1 }}>
                                    <CardContent>
                                        <Stack direction="row" justifyContent="space-between" alignItems="center" mb={2}>
                                            <Typography variant="h6">Recent Transactions</Typography>
                                            <Button size="small" onClick={() => navigate('/transactions')}>View All</Button>
                                        </Stack>
                                        <Divider sx={{ mb: 2 }} />
                                        <Box sx={{ overflowX: 'auto' }}>
                                            <Table size="small">
                                                <TableHead>
                                                    <TableRow>
                                                        <TableCell>Reference</TableCell>
                                                        <TableCell>Type</TableCell>
                                                        <TableCell align="right">Amount</TableCell>
                                                        <TableCell>Status</TableCell>
                                                        <TableCell>Date</TableCell>
                                                    </TableRow>
                                                </TableHead>
                                                <TableBody>
                                                    {selectedTransactions.map((transaction) => (
                                                        <TableRow key={transaction.id} hover>
                                                            <TableCell>{transaction.reference}</TableCell>
                                                            <TableCell>{transaction.type}</TableCell>
                                                            <TableCell align="right">{formatCurrency(transaction.value)}</TableCell>
                                                            <TableCell><Chip label={transaction.status} size="small" color={transaction.status === 'success' ? 'success' : transaction.status === 'pending' ? 'warning' : 'error'} /></TableCell>
                                                            <TableCell>{formatDate(transaction.created_at)}</TableCell>
                                                        </TableRow>
                                                    ))}
                                                </TableBody>
                                            </Table>
                                        </Box>
                                    </CardContent>
                                </Card>
                            </Stack>
                        </Grid>

                        <Grid item xs={12} md={5}>
                            <Stack spacing={3}>
                                <Card sx={{ borderRadius: 1 }}>
                                    <CardContent>
                                        <Typography variant="h6" mb={2}>Hierarchy Under This Vendor</Typography>
                                        <Paper variant="outlined" sx={{ maxHeight: 340, overflow: 'auto', borderRadius: 1 }}>
                                            {renderTree([selectedOrg])}
                                        </Paper>
                                    </CardContent>
                                </Card>

                                {/* <Card sx={{ borderRadius: 1 }}>
                                    <CardContent>
                                        <Stack direction="row" justifyContent="space-between" alignItems="center" mb={2}>
                                            <Typography variant="h6">Role Bindings ({selectedBindings.length})</Typography>
                                            <Button size="small" variant="contained" startIcon={<PersonAdd />} onClick={() => setBindingDialogOpen(true)}>
                                                Assign
                                            </Button>
                                        </Stack>
                                        <Stack spacing={1.5}>
                                            {selectedBindings.map((binding) => (
                                                <Paper key={binding.id} variant="outlined" sx={{ p: 1.5, borderRadius: 1 }}>
                                                    <Stack direction="row" justifyContent="space-between" spacing={2}>
                                                        <Box>
                                                            <Typography variant="subtitle2">{binding.user_email}</Typography>
                                                            <Stack direction="row" spacing={1} mt={0.5}>
                                                                <Chip label={levelLabel(binding.role)} size="small" color={levelColor(binding.role)} />
                                                                <Chip label={binding.status} size="small" color={statusColor(binding.status)} />
                                                            </Stack>
                                                        </Box>
                                                        <Stack direction="row" spacing={1}>
                                                            <Button size="small" onClick={() => toggleBindingStatus(binding.id)}>
                                                                {binding.status === 'active' ? 'Suspend' : 'Activate'}
                                                            </Button>
                                                            <IconButton size="small" color="error" onClick={() => removeBinding(binding.id)}>
                                                                <Delete fontSize="small" />
                                                            </IconButton>
                                                        </Stack>
                                                    </Stack>
                                                </Paper>
                                            ))}
                                        </Stack>
                                    </CardContent>
                                </Card> */}

                                <Card sx={{ borderRadius: 1 }}>
                                    <CardContent>
                                        <Stack direction="row" justifyContent="space-between" alignItems="center" mb={2}>
                                            <Typography variant="h6">Wallet Summary</Typography>
                                            <Button size="small" variant="outlined" onClick={() => navigate('/tenants/wallets')}>Manage</Button>
                                        </Stack>
                                        <Paper variant="outlined" sx={{ p: 2, mb: 2, bgcolor: 'primary.light', color: 'primary.contrastText', borderRadius: 1 }}>
                                            <Typography variant="caption">Total Balance</Typography>
                                            <Typography variant="h5" fontWeight={700}>{formatCurrency(totalSelectedWalletBalance)}</Typography>
                                        </Paper>
                                        <Stack spacing={1.5}>
                                            {selectedWallets.map((wallet) => (
                                                <Stack key={wallet.id} direction="row" justifyContent="space-between" spacing={2}>
                                                    <Box>
                                                        <Typography variant="subtitle2">{wallet.key}</Typography>
                                                        <Stack direction="row" spacing={1} mt={0.5}>
                                                            <Chip label={wallet.type} size="small" />
                                                            {wallet.role && <Chip label={wallet.role} size="small" variant="outlined" />}
                                                        </Stack>
                                                    </Box>
                                                    <Typography variant="subtitle2">{formatCurrency(wallet.balance)}</Typography>
                                                </Stack>
                                            ))}
                                        </Stack>
                                    </CardContent>
                                </Card>
                            </Stack>
                        </Grid>
                    </Grid>
                </Box>
            )}

            <Dialog open={createDialogOpen} onClose={() => setCreateDialogOpen(false)} maxWidth="sm" fullWidth>
                <DialogTitle>Create Organisation</DialogTitle>
                <DialogContent>
                    <Stack spacing={2} mt={1}>
                        <TextField label="Organisation Name" value={formState.name} onChange={(event) => setFormState((prev) => ({ ...prev, name: event.target.value }))} fullWidth autoFocus />
                        <TextField label="Location" value={formState.location} onChange={(event) => setFormState((prev) => ({ ...prev, location: event.target.value }))} fullWidth />
                        <TextField label="Manager Email" value={formState.manager_email} onChange={(event) => setFormState((prev) => ({ ...prev, manager_email: event.target.value }))} fullWidth />
                        <FormControl fullWidth>
                            <InputLabel>Level</InputLabel>
                            <Select value={formState.level} label="Level" onChange={(event) => setFormState((prev) => ({ ...prev, level: event.target.value as OrgLevel }))}>
                                {LEVEL_OPTIONS.map((option) => <MenuItem key={option.value} value={option.value}>{option.label}</MenuItem>)}
                            </Select>
                        </FormControl>
                    </Stack>
                </DialogContent>
                <DialogActions>
                    <Button onClick={() => setCreateDialogOpen(false)}>Cancel</Button>
                    <Button variant="contained" startIcon={<Add />} onClick={handleCreateSubmit}>Create</Button>
                </DialogActions>
            </Dialog>

            <Dialog open={editDialogOpen} onClose={() => setEditDialogOpen(false)} maxWidth="sm" fullWidth>
                <DialogTitle>Edit Organisation</DialogTitle>
                <DialogContent>
                    <Stack spacing={2} mt={1}>
                        <TextField label="Organisation Name" value={formState.name} onChange={(event) => setFormState((prev) => ({ ...prev, name: event.target.value }))} fullWidth autoFocus />
                        <TextField label="Location" value={formState.location} onChange={(event) => setFormState((prev) => ({ ...prev, location: event.target.value }))} fullWidth />
                        <TextField label="Manager Email" value={formState.manager_email} onChange={(event) => setFormState((prev) => ({ ...prev, manager_email: event.target.value }))} fullWidth />
                        <FormControl fullWidth>
                            <InputLabel>Status</InputLabel>
                            <Select value={formState.status} label="Status" onChange={(event) => setFormState((prev) => ({ ...prev, status: event.target.value as OrgStatus }))}>
                                {STATUS_OPTIONS.map((option) => <MenuItem key={option.value} value={option.value}>{option.label}</MenuItem>)}
                            </Select>
                        </FormControl>
                    </Stack>
                </DialogContent>
                <DialogActions>
                    <Button onClick={() => setEditDialogOpen(false)}>Cancel</Button>
                    <Button variant="contained" startIcon={<SaveIcon />} onClick={handleEditSubmit}>Save</Button>
                </DialogActions>
            </Dialog>

            <Dialog open={deleteDialogOpen} onClose={() => setDeleteDialogOpen(false)} maxWidth="xs" fullWidth>
                <DialogTitle>
                    <Stack direction="row" spacing={1} alignItems="center">
                        <WarningIcon color="error" />
                        <Typography color="error">Delete Organisation</Typography>
                    </Stack>
                </DialogTitle>
                <DialogContent>
                    <Typography>Delete <strong>{selectedOrg?.name}</strong> and its child organisations?</Typography>
                </DialogContent>
                <DialogActions>
                    <Button onClick={() => setDeleteDialogOpen(false)}>Cancel</Button>
                    <Button variant="contained" color="error" startIcon={<Delete />} onClick={handleDeleteSubmit}>Delete</Button>
                </DialogActions>
            </Dialog>

            <Dialog open={bindingDialogOpen} onClose={() => setBindingDialogOpen(false)} maxWidth="sm" fullWidth>
                <DialogTitle>Assign User to Organisation</DialogTitle>
                <DialogContent>
                    <Stack spacing={2} mt={1}>
                        <TextField label="User Email" value={bindingForm.user_email} onChange={(event) => setBindingForm((prev) => ({ ...prev, user_email: event.target.value }))} fullWidth />
                        <FormControl fullWidth>
                            <InputLabel>User Role</InputLabel>
                            <Select value={bindingForm.role} label="User Role" onChange={(event) => setBindingForm((prev) => ({ ...prev, role: event.target.value as OrgLevel }))}>
                                {LEVEL_OPTIONS.map((option) => <MenuItem key={option.value} value={option.value}>{option.label}</MenuItem>)}
                            </Select>
                        </FormControl>
                    </Stack>
                </DialogContent>
                <DialogActions>
                    <Button onClick={() => setBindingDialogOpen(false)}>Cancel</Button>
                    <Button variant="contained" startIcon={<PersonAdd />} onClick={handleBindingSubmit} disabled={!bindingForm.user_email.trim()}>Assign</Button>
                </DialogActions>
            </Dialog>

            <Snackbar open={Boolean(success)} autoHideDuration={3000} onClose={() => setSuccess(null)} anchorOrigin={{ vertical: 'top', horizontal: 'right' }}>
                <Paper sx={{ p: 1.5, bgcolor: 'success.main', color: 'success.contrastText' }}>
                    {success}
                </Paper>
            </Snackbar>
        </Box>
    );
};
